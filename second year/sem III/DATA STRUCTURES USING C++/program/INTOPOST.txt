#include<iostream.h>
#include<conio.h>
#define max 50
class   stack
{
 public:
   int item[max],top;
   void create(stack*);
   void push(stack*,int);
   int pop(stack*);
};
void  stack::create(stack *p)
{
   p->top=-1;
}
void stack::push(stack *p,int n)
{
    if(p->top==max-1)
    {
       cout<<"\nStack overflows..";
    }
    else
    {
       ++p->top;
       p->item[p->top]=n;
    }
}
int   stack::pop(stack *p)
{
    if(p->top==-1)
    {
       cout<<"\nStack underflows..";
       return(0);
    }
    else
    {
       return(p->item[p->top--]);
    }
}
