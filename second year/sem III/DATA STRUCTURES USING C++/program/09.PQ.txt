#define  max  5
class    queue
{
  int  item[max],pri[max];
  int  front,rear;
  public:
    void    create(queue*);
    void    insert(queue*,int,int);
    int     remove(queue*);
    void    display(queue*);
};
void     queue::create(queue   *p)
{
   p->front=p->rear=-1;
   cout<<"\nQueue is created..";
}
void     queue::insert(queue  *p,int   x,int   pr)
{
    if(p->rear==max-1)   //full
    {
       cout<<"\nQueue is overflows..";
    }
    else
    {
       ++p->rear;
       p->item[p->rear]=x;
       p->pri[p->rear]=pr;
       cout<<"\nElement is inserted..";
    }
}

int    queue::remove(queue   *p)
{
    int mx,i,pos=0,z;
     if(p->front==p->rear)   //empty
     {
	 cout<<"\nQueue is underflows..";
	 return(0);
     }
     else
     {
	 mx=p->pri[0];
	 for(i=p->front+1;i<=p->rear;i++)
	 {
	    if(mx<p->pri[i])
	    {
	       mx=p->pri[i];
	       pos=i;
	    }
	 }
	 z=p->item[pos];
	 for(i=pos;i<=p->rear;i++)
	 {
	      p->item[i]=p->item[i+1];
	      p->pri[i]=p->pri[i+1];
	 }
	 --p->rear;
	 return(z);
     }
}

void     queue::display(queue  *p)
{
   int   i;
   cout<<"\nElements: ";
   for(i=p->front+1;i<=p->rear;i++)
   {
      cout<<"\t"<<p->item[i];
   }
   cout<<"\nPriority: ";
   for(i=p->front+1;i<=p->rear;i++)
   {
      cout<<"\t"<<p->pri[i];
   }
}
