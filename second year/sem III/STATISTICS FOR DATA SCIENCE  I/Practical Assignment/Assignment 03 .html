# draw histogram to represent following data (unequal class width)

# class            f
# 50-100          40
# 100-150         55
# 150-200         60
# 200-300         54
# 300-400         48
# 400-550         40
# 550-750         30
# 750-1000        20


#program - TO draw histogram for unequal class-width
import numpy as np
import matplotlib.pyplot as plt
L=[50,100,150,200,300,400,550,750,1000] #class limit
f=[40,55,60,54,48,40,30,20] #frequency

#comput class-width
w=np.diff(L)

# comput freq.density
density=f/w
plt.bar(L[:-1],density,width=w,align='edge',color='red',edgecolor='black')
plt.title("Histogram for unequal classes")
plt.xlabel("class_limits")
plt.ylabel("frequency density")
plt.show()

"""
Write a Python program to draw and display a histogram for the data given below:

Class (x)    Frequency (f)
0-5          12
5-10         18
10-15        27
15-20        15
20-25        8
25-30        4
"""

import matplotlib.pyplot as plt

# Define class limits and frequencies

f = [12, 18, 27, 15, 8, 4]

plt.ylabel("Frequency")

# 02.to_draw_less_than_ogive.py
import numpy as np
import matplotlib.pyplot as plt
limits=[50,60,70,80,90,100,110,120,130]
f=[7,12,19,27,24,12,8,6]
lcf=np.cumsum(f) #to find lcf
new_lcf=np.insert(lcf,0,0)
plt.plot(limits,new_lcf,color='red',linestyle='-',marker='1')
plt.title("less than ogive")
plt.xlabel("class limits")
plt.ylabel("lcf")
plt.grid()
plt.show()

# 03. To draw greater than ogive for the following data
"""
class     f
25-30     5
30-35     11
35-40     17
40-45     23
45-50     33
50-55     29
55-60     17
60-65     8
65-70     3
"""
import numpy as np
import matplotlib.pyplot as plt

limits = [25,30,35,40,45,50,55,60,65,70]
F = [5,11,17,23,33,29,17,8,3]

revF = list(reversed(F))       # Frequency is reversed
lcf_rev = np.cumsum(revF)     # Cumulative frequency of reversed frequency

new_lcf = np.insert(lcf_rev, 0, 0)   # Zero is inserted
gcf = list(reversed(new_lcf))        # To obtain GCF

plt.plot(limits, gcf, color='red', linestyle='--', marker='1')
plt.xlabel("Class Limit")
plt.ylabel("GCF")
plt.title("Greater Than Ogive")
plt.grid()
plt.show()

#4) To draw Less Than and Greater Than Ogive

import numpy as np
import matplotlib.pyplot as plt

limits = [25,30,35,40,45,50,55,60,65,70]
F = [5,11,17,23,33,29,17,8,3]

# ---------------- Less Than Ogive ----------------

lcf = np.cumsum(F)
new_lcf = np.insert(lcf, 0, 0)

# ---------------- Greater Than Ogive ----------------

revF = list(reversed(F))
gcf_rev = np.cumsum(revF)
new_gcf = np.insert(gcf_rev, 0, 0)
gcf = list(reversed(new_gcf))

# ---------------- Plot ----------------

plt.plot(limits, new_lcf, color='blue',linestyle='--', marker='o', label='Less Than Ogive')

plt.plot(limits, gcf, color='red',linestyle='--', marker='1', label='Greater Than Ogive')

plt.xlabel("Class Limit")
plt.ylabel("Cumulative Frequency")
plt.title("Less Than and Greater Than Ogive")

plt.legend()
plt.grid()
plt.show()

# To draw pie chart

import matplotlib.pyplot as plt
cost=[2000,3000,1500,1800,2500,3500]
explode = [0, 0, 0, 0, 0, 0.1]
items=["steel","sand","bricks","timber","cement","labour"]
